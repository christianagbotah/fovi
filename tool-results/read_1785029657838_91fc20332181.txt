    72→function TerminalInner() {
    73→  const router = useRouter();
    74→  const { symbols, lastTickAt } = useMarket();
    75→  const dataLive = !!lastTickAt && Date.now() - lastTickAt < 6000;
    76→  const { authenticated, loading: authLoading, user, needsCompliance, refresh: refreshCompliance } = useCompliance();
    77→
    78→  const [account, setAccount] = React.useState<LinkedAccount | null>(null);
    79→  const [config, setConfig] = React.useState<BotConfig | null>(null);
    80→  const [stats, setStats] = React.useState<Stats>(DEFAULT_STATS);
    81→  const [positions, setPositions] = React.useState<Position[]>([]);
    82→  const [history, setHistory] = React.useState<ClosedTrade[]>([]);
    83→  const [decisions, setDecisions] = React.useState<AiDecision[]>([]);
    84→  const [selected, setSelected] = React.useState("BTC/USD");
    85→  const [loaded, setLoaded] = React.useState(false);
    86→  const [killSwitchOpen, setKillSwitchOpen] = React.useState(false);
    87→
    88→  const livePrices = React.useMemo(() => {
    89→    const map: Record<string, number> = {};
    90→    for (const s of symbols) map[s.symbol] = s.price;
    91→    return map;
    92→  }, [symbols]);
    93→
    94→  // Live unrealized P&L from socket prices (snappy UI)
    95→  const liveUnrealized = React.useMemo(() => {
    96→    return positions.reduce((acc, p) => {
    97→      const live = livePrices[p.symbol] ?? p.currentPrice;
    98→      const pnl =
    99→        p.side === "LONG"
   100→          ? (live - p.entryPrice) * p.size
   101→          : (p.entryPrice - live) * p.size;
   102→      return acc + pnl;
   103→    }, 0);
   104→  }, [positions, livePrices]);
   105→
   106→  async function refreshAll() {
   107→    const [statsRes, posRes, decRes, histRes, accRes, cfgRes] = await Promise.all([
   108→      fetch("/api/stats").then((r) => r.json()),
   109→      fetch("/api/positions").then((r) => r.json()),
   110→      fetch("/api/decisions").then((r) => r.json()),
   111→      fetch("/api/history").then((r) => r.json()),
   112→      fetch("/api/account").then((r) => r.json()),
   113→      fetch("/api/bot").then((r) => r.json()),
   114→    ]);
   115→    setStats(statsRes ?? DEFAULT_STATS);
   116→    setPositions(posRes?.positions ?? []);
   117→    setDecisions(decRes?.decisions ?? []);
   118→    setHistory(histRes?.trades ?? []);
   119→    setAccount(accRes?.account ?? null);
   120→    setConfig(cfgRes?.config ?? null);
   121→    setLoaded(true);
   122→  }
   123→
   124→  // Initial load
   125→  React.useEffect(() => {
   126→    refreshAll();
   127→  }, []);
   128→
   129→  // Bot tick loop — always runs (manages TP/SL on open positions; runs AI when enabled + throttled)
   130→  React.useEffect(() => {
   131→    let cancelled = false;
   132→    let timer: ReturnType<typeof setTimeout>;
   133→
   134→    async function tick() {
   135→      try {
   136→        const res = await fetch("/api/bot/tick", { method: "POST" });
   137→        const data = await res.json();
   138→        if (cancelled) return;
   139→        if (data?.closedPositions?.length) {
   140→          for (const c of data.closedPositions) {
   141→            toast(
   142→              `${c.symbol} closed (${c.reason}): ${c.pnl >= 0 ? "+" : ""}$${c.pnl.toFixed(2)}`,
   143→              {
   144→                description:
   145→                  c.reason === "TP"
   146→                    ? "Take profit hit"
   147→                    : c.reason === "SL"
   148→                      ? "Stop loss hit"
   149→                      : "Position exited",
   150→              },
   151→            );
   152→          }
   153→        }
   154→        if (data?.decision && data.decision.executed && data.decision.action !== "HOLD") {
   155→          toast.success(
   156→            `AI ${data.decision.action} ${data.decision.symbol} (${data.decision.confidence}% confidence)`,
   157→          );
   158→        }
   159→        await refreshAll();
   160→      } catch {
   161→        // silent — will retry
   162→      } finally {
   163→        if (!cancelled) timer = setTimeout(tick, 15000); // 15s — TP/SL still managed promptly, LLM is throttled server-side
   164→      }
   165→    }
   166→
   167→    timer = setTimeout(tick, 6000);
   168→    return () => {
   169→      cancelled = true;
   170→      clearTimeout(timer);
   171→    };
   172→  }, []);
   173→
   174→  // Re-render safe: refresh stats periodically too (independent of tick)
   175→  React.useEffect(() => {
   176→    const id = setInterval(refreshAll, 8000);
   177→    return () => clearInterval(id);
   178→  }, []);
   179→
   180→  const hasAccount = !!account;
   181→  const botActive = !!config?.enabled;
   182→
   183→  return (
   184→    <div className="flex min-h-screen flex-col overflow-x-clip bg-background">
   185→      {/* Header */}
   186→      <header className="sticky top-0 z-30 border-b bg-background/80 backdrop-blur supports-[backdrop-filter]:bg-background/60">
   187→        <div className="mx-auto flex min-h-14 max-w-7xl flex-nowrap items-center gap-x-1 px-3 py-1.5 sm:min-h-16 sm:flex-wrap sm:gap-x-3 sm:gap-y-2 sm:px-4 sm:py-2">
   188→          <div className="flex min-w-0 items-center gap-1.5 sm:gap-2">
   189→            <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-primary text-primary-foreground sm:h-9 sm:w-9">
   190→              <Bot className="h-4 w-4 sm:h-5 sm:w-5" />
   191→            </div>
   192→            <div className="leading-tight">
   193→              <div className="flex items-center gap-2">
   194→                <span className="text-sm font-bold tracking-tight sm:text-base">Fovi</span>
   195→                <Badge variant="secondary" className="hidden text-[10px] uppercase sm:inline-flex">
   196→                  AI Terminal
   197→                </Badge>
   198→              </div>
   199→              <span className="hidden text-[11px] text-muted-foreground sm:block">
   200→                Autonomous trading strategist
   201→              </span>
   202→            </div>
   203→          </div>
   204→
   205→          <div className="hidden items-center gap-1.5 md:flex">
   206→            <Badge
   207→              variant="outline"
   208→              className="gap-1 rounded-md"
   209→              title={dataLive ? "Market feed live" : "Market feed disconnected"}
   210→            >
   211→              {dataLive ? (
   212→                <Wifi className="h-3 w-3 text-emerald-500" />
   213→              ) : (
   214→                <WifiOff className="h-3 w-3 text-red-500" />
   215→              )}
   216→              <span className="text-[10px] uppercase">{dataLive ? "Live" : "Offline"}</span>
   217→            </Badge>

   220→                <Radio className="h-3 w-3 animate-pulse" />
   221→                <span className="text-[10px] uppercase">AI Active</span>
   222→              </Badge>
   223→            )}
   224→          </div>
   225→
   226→          {/* Compact live/AI badges shown inline on mobile (sm-md) when space is tight */}
   227→          <div className="ml-1 flex items-center gap-1 sm:ml-3 md:hidden">
   228→            <Badge
   229→              variant="outline"
   230→              className="gap-1 rounded-md px-1.5 py-0"
   231→              title={dataLive ? "Market feed live" : "Market feed disconnected"}

   293→      />
   294→
   295→      {/* Main */}

   297→        {authLoading ? (
   298→          <div className="flex h-64 items-center justify-center text-muted-foreground">
   299→            <Loader2 className="mr-2 h-5 w-5 animate-spin" />
   300→            Authenticating…
   301→          </div>

   303→          <AuthPrompt onSignIn={() => router.push("/auth/signin")} />
   304→        ) : needsCompliance && user ? (
   305→          <>
   306→            <ComplianceGate user={user} onAccepted={refreshCompliance} />
   307→            <div className="flex h-64 items-center justify-center text-muted-foreground">
   308→              <Loader2 className="mr-2 h-5 w-5 animate-spin" />
   309→              Waiting for risk acknowledgment…
   310→            </div>
   311→          </>
   312→        ) : !loaded ? (
   313→          <div className="flex h-64 items-center justify-center text-muted-foreground">
   314→            <Loader2 className="mr-2 h-5 w-5 animate-spin" />
   315→            Loading terminal…
   316→          </div>

   321→            <StatCards

   325→                equity: stats.balance + liveUnrealized,
   326→              }}
   327→            />
   328→
   329→            <div className="grid items-stretch gap-4 sm:gap-5 lg:grid-cols-3">

   331→                <DecisionSummaryCard decisions={decisions} botActive={botActive} />
   332→              </div>
   333→              <BotPanel
   334→                config={config}
   335→                onConfigChanged={refreshAll}
   336→                hasAccount={hasAccount}
   337→                killSwitchActive={!!user?.killSwitch}
   338→              />
   339→            </div>
   340→
   341→            <Tabs defaultValue="overview" className="w-full min-w-0">
   342→              <TabsList className="flex h-9 w-full justify-start gap-1 overflow-x-auto rounded-lg p-1 sm:grid sm:w-auto sm:grid-flow-col sm:justify-center sm:overflow-hidden [&::-webkit-scrollbar]:hidden [scrollbar-width:none]">
   343→                <TabsTrigger value="overview" className="shrink-0 gap-1.5 px-3">

   350→                  {stats.openCount > 0 && (

   352→                      {stats.openCount}
   353→                    </Badge>

   355→                </TabsTrigger>

   357→                  <Bot className="h-4 w-4" />

   359→                </TabsTrigger>

   668→function KillSwitchDialog({
   669→  open,
   670→  onOpenChange,
   671→  active,
   672→  onTriggered,
   673→}: {
   674→  open: boolean;
   675→  onOpenChange: (v: boolean) => void;
   676→  active: boolean;
   677→  onTriggered: () => void;
   678→}) {
   679→  const [loading, setLoading] = React.useState(false);
   680→
   681→  async function toggle(next: boolean) {
   682→    setLoading(true);
   683→    try {
   684→      const res = await fetch("/api/auth/kill-switch", {
   685→        method: "POST",
   686→        headers: { "Content-Type": "application/json" },
   687→        body: JSON.stringify({ enabled: next }),
   688→      });
   689→      if (!res.ok) throw new Error("Failed");
   690→      toast.success(
   691→        next
   692→          ? "Kill switch ON — all AI trading halted."
   693→          : "Kill switch off — you may re-enable the bot.",
   694→      );
   695→      onTriggered();
   696→      onOpenChange(false);
   697→    } catch {
   698→      toast.error("Could not toggle kill switch.");
   699→    } finally {
   700→      setLoading(false);
   701→    }
   702→  }
   703→
   704→  return (
   705→    <AlertDialog open={open} onOpenChange={onOpenChange}>
   706→      <AlertDialogContent>
   707→        <AlertDialogHeader>
   708→          <AlertDialogTitle className="flex items-center gap-2">
   709→            <OctagonAlert className="h-5 w-5 text-red-600" />
   710→            Emergency kill switch
   711→          </AlertDialogTitle>
   712→          <AlertDialogDescription>
   713→            {active
   714→              ? "The kill switch is currently ON. The AI bot cannot place new trades. Turn it off to resume (you will need to re-enable the bot manually)."
   715→              : "Activating the kill switch immediately halts the AI strategist and prevents any new trades from being placed. Open positions are NOT closed automatically — manage them manually. This is recorded in your audit log."}
   716→          </AlertDialogDescription>
   717→        </AlertDialogHeader>
   718→        <AlertDialogFooter>
   719→          <AlertDialogCancel>Cancel</AlertDialogCancel>
   720→          {active ? (
   721→            <AlertDialogAction
   722→              onClick={() => toggle(false)}
   723→              disabled={loading}
   724→              className="bg-emerald-600 hover:bg-emerald-700"
   725→            >
   726→              {loading && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   727→              Turn off
   728→            </AlertDialogAction>
   729→          ) : (
   730→            <AlertDialogAction
   731→              onClick={() => toggle(true)}
   732→              disabled={loading}
   733→              className="bg-red-600 hover:bg-red-700"
   734→            >
   735→              {loading && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   736→              Activate kill switch
   737→            </AlertDialogAction>
   738→          )}
   739→        </AlertDialogFooter>
   740→      </AlertDialogContent>
   741→    </AlertDialog>
   742→  );
   743→}
