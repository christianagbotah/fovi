   668→   668→function KillSwitchDialog({
   669→   669→  open,
   670→   670→  onOpenChange,
   671→   671→  active,
   672→   672→  onTriggered,
   673→   673→}: {
   674→   674→  open: boolean;
   675→   675→  onOpenChange: (v: boolean) => void;
   676→   676→  active: boolean;
   677→   677→  onTriggered: () => void;
   678→   678→}) {
   679→   679→  const [loading, setLoading] = React.useState(false);
   680→   680→
   681→   681→  async function toggle(next: boolean) {
   682→   682→    setLoading(true);
   683→   683→    try {
   684→   684→      const res = await fetch("/api/auth/kill-switch", {
   685→   685→        method: "POST",
   686→   686→        headers: { "Content-Type": "application/json" },
   687→   687→        body: JSON.stringify({ enabled: next }),
   688→   688→      });
   689→   689→      if (!res.ok) throw new Error("Failed");
   690→   690→      toast.success(
   691→   691→        next
   692→   692→          ? "Kill switch ON — all AI trading halted."
   693→   693→          : "Kill switch off — you may re-enable the bot.",
   694→   694→      );
   695→   695→      onTriggered();
   696→   696→      onOpenChange(false);
   697→   697→    } catch {
   698→   698→      toast.error("Could not toggle kill switch.");
   699→   699→    } finally {
   700→   700→      setLoading(false);
   701→   701→    }
   702→   702→  }
   703→   703→
   704→   704→  return (
   705→   705→    <AlertDialog open={open} onOpenChange={onOpenChange}>
   706→   706→      <AlertDialogContent>
   707→   707→        <AlertDialogHeader>
   708→   708→          <AlertDialogTitle className="flex items-center gap-2">
   709→   709→            <OctagonAlert className="h-5 w-5 text-red-600" />
   710→   710→            Emergency kill switch
   711→   711→          </AlertDialogTitle>
   712→   712→          <AlertDialogDescription>
   713→   713→            {active
   714→   714→              ? "The kill switch is currently ON. The AI bot cannot place new trades. Turn it off to resume (you will need to re-enable the bot manually)."
   715→   715→              : "Activating the kill switch immediately halts the AI strategist and prevents any new trades from being placed. Open positions are NOT closed automatically — manage them manually. This is recorded in your audit log."}
   716→   716→          </AlertDialogDescription>
   717→   717→        </AlertDialogHeader>
   718→   718→        <AlertDialogFooter>
   719→   719→          <AlertDialogCancel>Cancel</AlertDialogCancel>
   720→   720→          {active ? (
   721→   721→            <AlertDialogAction
   722→   722→              onClick={() => toggle(false)}
   723→   723→              disabled={loading}
   724→   724→              className="bg-emerald-600 hover:bg-emerald-700"
   725→   725→            >
   726→   726→              {loading && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   727→   727→              Turn off
   728→   728→            </AlertDialogAction>
   729→   729→          ) : (
   730→   730→            <AlertDialogAction
   731→   731→              onClick={() => toggle(true)}
   732→   732→              disabled={loading}
   733→   733→              className="bg-red-600 hover:bg-red-700"
   734→   734→            >
   735→   735→              {loading && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   736→   736→              Activate kill switch
   737→   737→            </AlertDialogAction>
   738→   738→          )}
   739→   739→        </AlertDialogFooter>
   740→   740→      </AlertDialogContent>
   741→   741→    </AlertDialog>
   742→   742→  );
   743→   743→}
