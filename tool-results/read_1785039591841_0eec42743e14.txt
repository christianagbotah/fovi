   391→// ============================================================
   392→// Simple Markdown renderer (no external dep needed)
   393→// ============================================================
   394→function MarkdownContent({ content }: { content: string }) {
   395→  const lines = content.split('\n');
   396→  return (
   397→    <>
   398→      {lines.map((line, j) => {
   399→        if (!line.trim()) return <br key={j} />;
   400→        if (line.startsWith('### ')) return <h4 key={j} className="font-bold text-sm mt-3 mb-1">{line.slice(4)}</h4>;
   401→        if (line.startsWith('## ')) return <h3 key={j} className="font-bold text-base mt-3 mb-1">{line.slice(3)}</h3>;
   402→        if (line.startsWith('**') && line.endsWith('**')) return <p key={j} className="font-bold mt-2 first:mt-0">{line.replace(/\*\*/g, '')}</p>;
   403→        if (line.startsWith('- ') || line.startsWith('* ')) {
   404→          const text = line.slice(2);
   405→          // Handle inline bold
   406→          const parts = text.split(/(\*\*[^*]+\*\*)/);
   407→          return (
   408→            <p key={j} className="ml-2 flex items-start gap-1">
   409→              <span className="text-primary mt-0.5">•</span>
   410→              <span>{parts.map((part, k) =>
   411→                part.startsWith('**') && part.endsWith('**')
   412→                  ? <strong key={k}>{part.replace(/\*\*/g, '')}</strong>
   413→                  : part
   414→              )}</span>
   415→            </p>
   416→          );
   417→        }
   418→        // Inline formatting
   419→        const parts = line.split(/(\*\*[^*]+\*\*)/);
   420→        if (line.startsWith('*') && line.endsWith('*') && !line.includes(' ')) {
   421→          return <p key={j} className="text-xs opacity-60 mt-2 italic">{line.replace(/\*/g, '')}</p>;
   422→        }
   423→        return <p key={j}>{parts.map((part, k) =>
   424→          part.startsWith('**') && part.endsWith('**')
   425→            ? <strong key={k}>{part.replace(/\*\*/g, '')}</strong>
   426→            : part
   427→        )}</p>;
   428→      })}
   429→    </>
   430→  );
   431→}
